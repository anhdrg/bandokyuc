function boDau(s){return String(s||"").normalize("NFD").replace(/[̀-ͯ]/g,"").replace(/đ/g,"d").replace(/Đ/g,"D");}
 
function boThe(s){return String(s||"").replace(/<[^>]*>/g,"").replace(/\s+/g," ").trim();}
 
// Tách mô tả dài thành phần mở đầu + các mục (Tên gọi, Đặc điểm, Giá trị, Ý nghĩa, Truyền thuyết...)
function tachMoTa(html){
 
  const lines=String(html||"").replace(/<b\s*\/>/gi,"</b>").split(/\n+/).map(s=>s.trim()).filter(Boolean);
 
  const re=/^(?:<b>\s*)?((?:Tên gọi|Đặc điểm|Giá trị|Ý nghĩa|Truyền thuyết|Truyền thống|Câu chuyện|Thực hành)[^:<：]{0,90}?)\s*(?:<\/b>)?\s*\.?\s*[:：]\s*(?:<\/b>)?\s*([\s\S]*)$/;
 
  const intro=[],sections=[],seen=new Set();
 
  lines.forEach(line=>{
 
    const m=line.match(re);
 
    if(!m){
 
      if(sections.length)sections[sections.length-1].noiDung+="<br>"+line;
 
      else intro.push(line);
 
      return;
 
    }
 
    const noiDung=m[2].trim();
 
    const khoa=boThe(noiDung).slice(0,200);
 
    if(!noiDung||seen.has(khoa))return;   // bỏ đoạn bị lặp
 
    seen.add(khoa);

}

function napVideo(d){
  const box=document.getElementById("modalVideo");box.replaceChildren();
  const src=(d.video||"").trim();
  if(!src){box.append(ghiChuRong("Chừa chỗ bổ sung video tư liệu cho hồ sơ này."));return;}

  let youtubeId="";
  try{
    const url=new URL(src);
    const host=url.hostname.toLowerCase().replace(/^www\./,"").replace(/^m\./,"");
    if(host==="youtu.be") youtubeId=url.pathname.split("/")[1]||"";
    else if(host==="youtube.com"){
      const parts=url.pathname.split("/");
      youtubeId=url.pathname==="/watch"?url.searchParams.get("v")||"":
        (parts[1]==="shorts"||parts[1]==="embed"?parts[2]||"":"");
    }
  }catch(e){} // Đường dẫn tệp tương đối vẫn được phát bằng thẻ video.

  if(/^[A-Za-z0-9_-]{11}$/.test(youtubeId)){
    const frame=document.createElement("iframe");
    frame.src="https://www.youtube.com/embed/"+youtubeId;
    frame.title="Video tư liệu: "+d.ten;
    frame.allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share";
    frame.allowFullscreen=true;
    frame.referrerPolicy="strict-origin-when-cross-origin";
    box.append(frame);
    const link=document.createElement("a");
    link.href="https://www.youtube.com/watch?v="+youtubeId;
    link.target="_blank";link.rel="noopener noreferrer";
    link.textContent="Xem video trên YouTube";
    box.append(link);
    return;
  }
  const v=document.createElement("video");v.controls=true;v.preload="metadata";
  v.onerror=()=>{box.replaceChildren(ghiChuRong("Chưa tìm thấy tệp video hoặc định dạng chưa được hỗ trợ."));};
  v.src=src;box.append(v);
}

function chonAnhBinhLuan(input){

  const loi=[];

  [...input.files].forEach(file=>{

    if(!/^image\/(jpeg|png|webp)$/.test(file.type)){loi.push(file.name+": chỉ nhận ảnh JPG, PNG, WEBP");return;}

    if(file.size>BL_TOI_DA_MB*1024*1024){loi.push(file.name+": lớn hơn "+BL_TOI_DA_MB+" MB");return;}

    if(anhBinhLuanChon.length>=BL_TOI_DA_ANH){loi.push(file.name+": chỉ được đính kèm tối đa "+BL_TOI_DA_ANH+" ảnh");return;}

    anhBinhLuanChon.push({file,url:URL.createObjectURL(file)});

  });

  input.value="";

  veXemTruocAnhBinhLuan();

  if(loi.length)alert("⚠️ Một số ảnh chưa được thêm:\n\n"+loi.join("\n"));

}
